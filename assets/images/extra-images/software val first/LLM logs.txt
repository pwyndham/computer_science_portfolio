/**
     * Adds a blind BCC recipient to the email. The email address will
	also be used as the personal name. The name will be encoded by
	the charset of
     * {@link #setCharset(String)}. If it is not set, it will be encoded
	using the Java platform's default charset (UTF-16) if it contains
	non-ASCII characters;
     * otherwise, it is used as is.
     *
     * @param email A String.
     * @return An Email.
     * @throws EmailException Indicates an invalid email address
     * @since 1.0
     */
    public Email addBcc(final String email) throws EmailException {
        return addBcc(email, null);
    }

    /**
     * Adds an array of blind BCC recipients to the email. The email 
	addresses will also be used as the personal name. The names 
	will be encoded by the charset
     * of {@link #setCharset(String)}. If it is not set, it will be 
	encoded using the Java platform's default charset (UTF-16) 
	if it contains non-ASCII
     * characters; otherwise, it is used as is.
     *
     * @param emails A String array.
     * @return An Email.
     * @throws EmailException Indicates an invalid email address
     * @since 1.3
     */
    public Email addBcc(final String... emails) throws EmailException {
        EmailException.checkNonEmpty(emails, () -> "BCC list invalid.");
        for (final String email : emails) {
            addBcc(email, null);
        }
        return this;
    }

    /**
     * Adds a blind BCC recipient to the email using the specified address
	and the specified personal name. The name will be encoded by the 
	charset of
     * {@link #setCharset(String)}. If it is not set, it will be encoded 
	using the Java platform's default charset (UTF-16) if it contains 
	non-ASCII characters;
     * otherwise, it is used as is.
     *
     * @param email A String.
     * @param name  A String.
     * @return An Email.
     * @throws EmailException Indicates an invalid email address
     * @since 1.0
     */
    public Email addBcc(final String email, final String name) throws 
	EmailException {
        return addBcc(email, name, charset);
    }

    /**
     * Adds a blind BCC recipient to the email using the specified address,
	personal name, and charset encoding for the name.
     *
     * @param email   A String.
     * @param name    A String.
     * @param charset The charset to encode the name with.
     * @return An Email.
     * @throws EmailException Indicates an invalid email address
     * @since 1.1
     */
    public Email addBcc(final String email, final String name, final String
	charset) throws EmailException {
        bccList.add(createInternetAddress(email, name, charset));
        return this;
    }

package org.apache.commons.mail2.javax;

/**
 * Unit tests for org.apache.commons.mail2.javax.Email class.
 * 
 * Tests the required 8 methods:
 *  - addBcc(String)
 *  - addHeader(String, String)
 *  - addReplyTo(String, String)
 *  - buildMimeMessage()
 *  - send()
 *  - getMailSession()
 *  - getSentDate()
 *  - updateContentType(String)
 */
class EmailTest {

    private SimpleEmail testEmail;

    @BeforeEach
    void setUp() {
        testEmail = new SimpleEmail();
    }

    @Test
    @DisplayName("Test addBcc() with valid email")
    void testAddBcc() throws EmailException {
        testEmail.addBcc("a@b.com");
        assertFalse(testEmail.getBccAddresses().isEmpty(), "BCC list should 
	not be empty after adding address");
        assertEquals("a@b.com", testEmail.getBccAddresses().get(0).getAddress());
    }

    @Test
    @DisplayName("Test addBcc() with invalid email throws EmailException")
    void testAddBccInvalidEmail() {
        assertThrows(EmailException.class, () -> testEmail.addBcc("invalid-email"));
    }

    @Test
    @DisplayName("Test addHeader() stores header correctly")
    void testAddHeader() {
        testEmail.addHeader("X-Test-Header", "Value123");
        assertTrue(testEmail.getHeaders().containsKey("X-Test-Header"));
        assertEquals("Value123", testEmail.getHeaders().get("X-Test-Header"));
    }

    @Test
    @DisplayName("Test addReplyTo() adds reply-to address correctly")
    void testAddReplyTo() throws EmailException {
        testEmail.addReplyTo("reply@example.com", "Reply User");
        assertFalse(testEmail.getReplyToAddresses().isEmpty());
        assertEquals("reply@example.com", testEmail.getReplyToAddresses().get(0).getAddress());
    }

    @Test
    @DisplayName("Test buildMimeMessage() runs without error after minimal setup")
    void testBuildMimeMessage() throws Exception {
        testEmail.setHostName("localhost");
        testEmail.setFrom("sender@example.com");
        testEmail.addTo("receiver@example.com");
        testEmail.setSubject("Test Subject");
        testEmail.setMsg("Test message body");

        assertDoesNotThrow(() -> testEmail.buildMimeMessage());
        assertNotNull(testEmail.getMimeMessage(), "MimeMessage should be built");
    }

    @Test
    @DisplayName("Test getMailSession() returns valid Session object")
    void testGetMailSession() throws EmailException {
        testEmail.setHostName("smtp.example.com");
        Session session = testEmail.getMailSession();
        assertNotNull(session);
        Properties props = session.getProperties();
        assertEquals("smtp.example.com", props.getProperty("mail.smtp.host"));
    }

    @Test
    @DisplayName("Test getSentDate() defaults and can be updated")
    void testGetSentDate() {
        Date sentDate = testEmail.getSentDate();
        assertNotNull(sentDate, "Sent date should default to non-null");
    }

    @Test
    @DisplayName("Test updateContentType() modifies internal content type")
    void testUpdateContentType() throws EmailException {
        testEmail.updateContentType("text/html; charset=UTF-8");
        assertTrue(testEmail.getContentType().contains("text/html"));
    }

    @Test
    @DisplayName("Test send() throws expected exception without SMTP setup")
    void testSend() {
        assertThrows(EmailException.class, testEmail::send);
    }
}
